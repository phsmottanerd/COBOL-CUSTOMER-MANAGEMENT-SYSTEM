def svg_start(width=1400, height=500):
    return f'''<svg xmlns="http://www.w3.org/2000/svg"
    width="{width}"
    height="{height}"
    viewBox="0 0 {width} {height}">

    <rect width="100%" height="100%" fill="{BLACK}"/>

    <rect
        x="12"
        y="12"
        width="{width - 24}"
        height="{height - 24}"
        rx="8"
        fill="none"
        stroke="{GREEN}"
        stroke-width="3"/>

    <rect
        x="28"
        y="28"
        width="{width - 56}"
        height="{height - 56}"
        rx="5"
        fill="none"
        stroke="{DARK_GREEN}"
        stroke-width="1"/>
'''


def svg_end():
    return "</svg>"


def text(x, y, value, size=22, weight=700,
         anchor="start", opacity=1.0):

    value = html.escape(value)

    return f'''
    <text
        x="{x}"
        y="{y}"
        fill="{GREEN}"
        opacity="{opacity}"
        font-family="{FONT}"
        font-size="{size}px"
        font-weight="{weight}"
        text-anchor="{anchor}">
        {value}
    </text>
    '''


def line(x1, y1, x2, y2, width=2):
    return f'''
    <line
        x1="{x1}"
        y1="{y1}"
        x2="{x2}"
        y2="{y2}"
        stroke="{GREEN}"
        stroke-width="{width}"/>
    '''


def box(x, y, width, height, radius=6):
    return f'''
    <rect
        x="{x}"
        y="{y}"
        width="{width}"
        height="{height}"
        rx="{radius}"
        fill="{BLACK}"
        stroke="{GREEN}"
        stroke-width="2"/>
    '''


def save_svg(filename, content):
    (ASSETS / filename).write_text(
        content,
        encoding="utf-8"
    )

Este projeto apresenta um sistema completo de **CRUD de Clientes desenvolvido em COBOL**, utilizando **GNUCOBOL**.
